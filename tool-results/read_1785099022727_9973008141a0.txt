    53→    53→class FrequencyDial(QWidget):
    54→    54→    """A digital frequency display + tuning buttons."""
    55→    55→    tune_requested = pyqtSignal(int)
    56→    56→
    57→    57→    def __init__(self, parent=None):
    58→    58→        super().__init__(parent)
    59→    59→        self.freq_hz = 96_900_000
    60→    60→        layout = QVBoxLayout(self)
    61→    61→        layout.setContentsMargins(0, 0, 0, 0)
    62→    62→
    63→    63→        # Big frequency display
    64→    64→        self.display = QLabel("96.900000 MHz")
    65→    65→        f = QFont("JetBrains Mono", 32)
    66→    66→        f.setWeight(QFont.Light)
    67→    67→        self.display.setFont(f)
    68→    68→        self.display.setStyleSheet("color: #5cd9ff; padding: 8px;")
    69→    69→        self.display.setAlignment(Qt.AlignCenter)
    70→    70→        layout.addWidget(self.display)
    71→    71→
    72→    72→        # Tuning buttons: -100k -10k -1k | entry | +1k +10k +100k
    73→    73→        row = QHBoxLayout()
    74→    74→        for step, label in [(-100_000, "−100k"), (-10_000, "−10k"), (-1_000, "−1k")]:
    75→    75→            btn = QPushButton(label)
    76→    76→            btn.clicked.connect(lambda _, s=step: self.tune_requested.emit(self.freq_hz + s))
    77→    77→            row.addWidget(btn)
    78→    78→        self.entry = QLineEdit()
    79→    79→        self.entry.setPlaceholderText("MHz")
    80→    80→        self.entry.setFixedWidth(120)
    81→    81→        self.entry.returnPressed.connect(self._on_entry)
    82→    82→        row.addWidget(self.entry)
    83→    83→        for step, label in [(1_000, "+1k"), (10_000, "+10k"), (100_000, "+100k")]:
    84→    84→            btn = QPushButton(label)
    85→    85→            btn.clicked.connect(lambda _, s=step: self.tune_requested.emit(self.freq_hz + s))
    86→    86→            row.addWidget(btn)
    87→    87→        layout.addLayout(row)
    88→    88→
    89→    89→    def set_frequency(self, freq_hz: int) -> None:
    90→    90→        self.freq_hz = int(freq_hz)
    91→    91→        self.display.setText(f"{self.freq_hz / 1e6:.6f} MHz")
    92→    92→
    93→    93→    def _on_entry(self) -> None:
    94→    94→        try:
    95→    95→            mhz = float(self.entry.text())
    96→    96→            self.tune_requested.emit(int(mhz * 1e6))
    97→    97→        except ValueError:
    98→    98→            pass

   700→   700→    def _on_test_sweep(self) -> None:
   701→   701→        """Diagnostic sweep: samples every frequency and shows ALL levels,
   702→   702→        regardless of threshold. Use this to figure out why 'Scan' returns 0.
   703→   703→        """
   704→   704→        if not self.gqrx.is_connected():
   705→   705→            QMessageBox.warning(self, "Not connected",
   706→   706→                                "Connect to Gqrx first, then run Test Sweep.")
   707→   707→            return
   708→   708→        if self.scanner.is_running():
   709→   709→            self.scanner.stop()
   710→   710→            return
   711→   711→        band_name = self.scan_band_combo.currentText()
   712→   712→        if band_name == "ALL BANDS":
   713→   713→            band_name = "FM Broadcast"  # test sweep is single-band only
   714→   714→        band = BANDS_BY_NAME.get(band_name)
   715→   715→        if not band:
   716→   716→            return
   717→   717→        # Run in a thread so the UI stays responsive
   718→   718→        import threading
   719→   719→        self.test_sweep_list.clear()
   720→   720→        self.scan_status.setText(f"Test sweeping {band.name}…")
   721→   721→        self.scan_progress.setValue(0)
   722→   722→
   723→   723→        def sweep():
   724→   724→            start_hz = int(band.start_mhz * 1e6)
   725→   725→            end_hz = int(band.end_mhz * 1e6)
   726→   726→            step_hz = int(band.step_khz * 1e3)
   727→   727→            n_steps = max(1, (end_hz - start_hz) // step_hz + 1)
   728→   728→            self.gqrx.set_modulation(band.modulation)
   729→   729→            time.sleep(0.1)
   730→   730→            results = []
   731→   731→            for i, f in enumerate(range(start_hz, end_hz + 1, step_hz)):
   732→   732→                if self.scanner._stop.is_set():
   733→   733→                    break
   734→   734→                self.gqrx.set_frequency(f)
   735→   735→                time.sleep(self.scanner.dwell_s)
   736→   736→                lvl = self.gqrx.get_signal_level()
   737→   737→                if lvl is None:
   738→   738→                    continue
   739→   739→                results.append((f, lvl))
   740→   740→                # Update progress UI (thread-safe via QMetaObject)
   741→   741→                from PyQt5.QtCore import QMetaObject, Qt as QQt, Q_ARG
   742→   742→                # Sort results and show top 50 strongest
   743→   743→                results.sort(key=lambda x: x[1], reverse=True)
   744→   744→                top = results[:50]
   745→   745→                QMetaObject.invokeMethod(self, "_refresh_test_sweep",
   746→   746→                                          QQt.QueuedConnection,
   747→   747→                                          Q_ARG(list, top))
   748→   748→                self.scan_progress.setValue(int((i + 1) / n_steps * 100))
   749→   749→            self.scan_status.setText(
   750→   750→                f"Test sweep done — {len(results)} freqs sampled. "
   751→   751→                f"Strongest: {results[0][0]/1e6:.4f} MHz @ {results[0][1]:.1f} dBFS"
   752→   752→                if results else "Test sweep done — no signal sampled (check antenna + gain)"
   753→   753→            )
   754→   754→
   755→   755→        threading.Thread(target=sweep, daemon=True, name="TestSweep").start()
   756→   756→
   757→   757→    # Slot invoked from the sweep thread to update the test-sweep list
   758→   758→    from PyQt5.QtCore import pyqtSlot
   759→   759→    @pyqtSlot(list)
   760→   760→    def _refresh_test_sweep(self, top_results: list) -> None:
   761→   761→        self.test_sweep_list.clear()
   762→   762→        for f, lvl in top_results:
   763→   763→            # Color-code: green for strong, yellow for medium, dim for weak
   764→   764→            if lvl > -50:
   765→   765→                color = "#5cffaa"
   766→   766→                tag = "STRONG"
   767→   767→            elif lvl > -65:
   768→   768→                color = "#ffd45c"
   769→   769→                tag = "medium"
   770→   770→            elif lvl > -80:
   771→   771→                color = "#8b96a7"
   772→   772→                tag = "weak"
   773→   773→            else:
   774→   774→                color = "#4a5266"
   775→   775→                tag = "noise"
   776→   776→            item = QListWidgetItem(f"{f/1e6:.4f} MHz · {lvl:.1f} dBFS · {tag}")
   777→   777→            item.setForeground(QColor(color))
   778→   778→            item.setData(Qt.UserRole, f)
   779→   779→            self.test_sweep_list.addItem(item)
