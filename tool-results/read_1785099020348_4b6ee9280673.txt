    53→class FrequencyDial(QWidget):
    54→    """A digital frequency display + tuning buttons."""
    55→    tune_requested = pyqtSignal(int)
    56→
    57→    def __init__(self, parent=None):
    58→        super().__init__(parent)
    59→        self.freq_hz = 96_900_000
    60→        layout = QVBoxLayout(self)
    61→        layout.setContentsMargins(0, 0, 0, 0)
    62→
    63→        # Big frequency display
    64→        self.display = QLabel("96.900000 MHz")
    65→        f = QFont("JetBrains Mono", 32)
    66→        f.setWeight(QFont.Light)
    67→        self.display.setFont(f)
    68→        self.display.setStyleSheet("color: #5cd9ff; padding: 8px;")
    69→        self.display.setAlignment(Qt.AlignCenter)
    70→        layout.addWidget(self.display)
    71→
    72→        # Tuning buttons: -100k -10k -1k | entry | +1k +10k +100k
    73→        row = QHBoxLayout()
    74→        for step, label in [(-100_000, "−100k"), (-10_000, "−10k"), (-1_000, "−1k")]:
    75→            btn = QPushButton(label)
    76→            btn.clicked.connect(lambda _, s=step: self.tune_requested.emit(self.freq_hz + s))
    77→            row.addWidget(btn)
    78→        self.entry = QLineEdit()
    79→        self.entry.setPlaceholderText("MHz")
    80→        self.entry.setFixedWidth(120)
    81→        self.entry.returnPressed.connect(self._on_entry)
    82→        row.addWidget(self.entry)
    83→        for step, label in [(1_000, "+1k"), (10_000, "+10k"), (100_000, "+100k")]:
    84→            btn = QPushButton(label)
    85→            btn.clicked.connect(lambda _, s=step: self.tune_requested.emit(self.freq_hz + s))
    86→            row.addWidget(btn)
    87→        layout.addLayout(row)
    88→
    89→    def set_frequency(self, freq_hz: int) -> None:
    90→        self.freq_hz = int(freq_hz)
    91→        self.display.setText(f"{self.freq_hz / 1e6:.6f} MHz")
    92→
    93→    def _on_entry(self) -> None:
    94→        try:
    95→            mhz = float(self.entry.text())
    96→            self.tune_requested.emit(int(mhz * 1e6))
    97→        except ValueError:
    98→            pass

   408→    # ----------------------------- signal wiring -----------------------------
   409→    def _wire_signals(self) -> None:
   410→        # Gqrx signals
   411→        self.gqrx.connected.connect(self._on_gqrx_connected)

   413→        self.gqrx.frequency_changed.connect(self._on_freq_changed)
   414→        self.gqrx.modulation_changed.connect(self._on_mod_changed)
   415→        self.gqrx.signal_level.connect(self._on_signal_level)
   416→        self.gqrx.error.connect(self._on_gqrx_error)
   417→
   418→        # Audio → player + recording
   419→        self.audio_receiver.chunk_ready.connect(self._on_audio_chunk)
   420→
   421→        # Spectrum → waterfall
   422→        self.spectrum_receiver.spectrum_ready.connect(
   423→            lambda data, c, s: self.waterfall.update_spectrum(data, c, s)
   424→        )
   425→
   426→        # Scanner

   428→        self.scanner.scan_progress.connect(self._on_scan_progress)
   429→        self.scanner.scan_progress_freq.connect(
   430→            lambda f: self.scan_status.setText(f"Scanning: {f/1e6:.4f} MHz")
   431→        )
   432→        # Live level readout — shows what the scanner is hearing in real time
   433→        self.scanner.scan_progress_level.connect(self._on_scan_progress_level)
   434→        self.scanner.station_found.connect(self._on_station_found)
   435→        self.scanner.scan_finished.connect(self._on_scan_finished)
   436→        self.scanner.scan_error.connect(self._on_scan_error)
   437→        self.scanner.noise_floor_calibrated.connect(self._on_noise_floor_calibrated)
   438→
   439→        # Recordings
   440→        self.recordings.recording_started.connect(self._on_recording_started)
   441→        self.recordings.recording_stopped.connect(self._on_recording_stopped)
   442→        self.recordings.chunk_recorded.connect(
   443→            lambda frames: self.rec_status.setText(f"● {frames / 48000:.1f}s")
   444→        )
   445→
   446→        # AI tagger
   447→        self.ai_tagger.tag_ready.connect(self._on_ai_tag_ready)
   448→        self.ai_tagger.tag_failed.connect(
   449→            lambda f, e: self.status.showMessage(f"AI tag failed for {f/1e6:.4f}: {e}", 3000)

   452→    def _apply_config(self) -> None:
   453→        self.dial.set_frequency(self.config.last_frequency_hz)
   454→        self.mod_combo.setCurrentText(self.config.last_modulation)
   455→        # Apply initial band context to waterfall
   456→        self._update_band_context(self.config.last_frequency_hz)
   457→        # Auto-start web server if enabled
   458→        if self.config.remote_access_enabled:
   459→            QTimer.singleShot(500, lambda: self.web_btn.click())

   461→    # ----------------------------- handlers -----------------------------
   462→    def _on_connect_clicked(self) -> None:
   463→        if self.gqrx.is_connected():
   464→            self.gqrx.disconnect()
   465→        else:
   466→            ok = self.gqrx.connect()
   467→            if not ok:

   700→    def _on_test_sweep(self) -> None:
   701→        """Diagnostic sweep: samples every frequency and shows ALL levels,
   702→        regardless of threshold. Use this to figure out why 'Scan' returns 0.
   703→        """
   704→        if not self.gqrx.is_connected():
   705→            QMessageBox.warning(self, "Not connected",
   706→                                "Connect to Gqrx first, then run Test Sweep.")
   707→            return
   708→        if self.scanner.is_running():
   709→            self.scanner.stop()
   710→            return
   711→        band_name = self.scan_band_combo.currentText()
   712→        if band_name == "ALL BANDS":
   713→            band_name = "FM Broadcast"  # test sweep is single-band only
   714→        band = BANDS_BY_NAME.get(band_name)
   715→        if not band:
   716→            return
   717→        # Run in a thread so the UI stays responsive
   718→        import threading
   719→        self.test_sweep_list.clear()
   720→        self.scan_status.setText(f"Test sweeping {band.name}…")
   721→        self.scan_progress.setValue(0)
   722→
   723→        def sweep():
   724→            start_hz = int(band.start_mhz * 1e6)
   725→            end_hz = int(band.end_mhz * 1e6)
   726→            step_hz = int(band.step_khz * 1e3)
   727→            n_steps = max(1, (end_hz - start_hz) // step_hz + 1)
   728→            self.gqrx.set_modulation(band.modulation)
   729→            time.sleep(0.1)
   730→            results = []
   731→            for i, f in enumerate(range(start_hz, end_hz + 1, step_hz)):
   732→                if self.scanner._stop.is_set():
   733→                    break
   734→                self.gqrx.set_frequency(f)
   735→                time.sleep(self.scanner.dwell_s)
   736→                lvl = self.gqrx.get_signal_level()
   737→                if lvl is None:
   738→                    continue
   739→                results.append((f, lvl))
   740→                # Update progress UI (thread-safe via QMetaObject)
   741→                from PyQt5.QtCore import QMetaObject, Qt as QQt, Q_ARG
   742→                # Sort results and show top 50 strongest
   743→                results.sort(key=lambda x: x[1], reverse=True)
   744→                top = results[:50]
   745→                QMetaObject.invokeMethod(self, "_refresh_test_sweep",
   746→                                          QQt.QueuedConnection,
   747→                                          Q_ARG(list, top))
   748→                self.scan_progress.setValue(int((i + 1) / n_steps * 100))
   749→            self.scan_status.setText(
   750→                f"Test sweep done — {len(results)} freqs sampled. "
   751→                f"Strongest: {results[0][0]/1e6:.4f} MHz @ {results[0][1]:.1f} dBFS"
   752→                if results else "Test sweep done — no signal sampled (check antenna + gain)"
   753→            )
   754→
   755→        threading.Thread(target=sweep, daemon=True, name="TestSweep").start()
   756→
   757→    # Slot invoked from the sweep thread to update the test-sweep list
   758→    from PyQt5.QtCore import pyqtSlot
   759→    @pyqtSlot(list)
   760→    def _refresh_test_sweep(self, top_results: list) -> None:
   761→        self.test_sweep_list.clear()
   762→        for f, lvl in top_results:
   763→            # Color-code: green for strong, yellow for medium, dim for weak
   764→            if lvl > -50:
   765→                color = "#5cffaa"
   766→                tag = "STRONG"
   767→            elif lvl > -65:
   768→                color = "#ffd45c"
   769→                tag = "medium"
   770→            elif lvl > -80:
   771→                color = "#8b96a7"
   772→                tag = "weak"
   773→            else:
   774→                color = "#4a5266"
   775→                tag = "noise"
   776→            item = QListWidgetItem(f"{f/1e6:.4f} MHz · {lvl:.1f} dBFS · {tag}")
   777→            item.setForeground(QColor(color))
   778→            item.setData(Qt.UserRole, f)
   779→            self.test_sweep_list.addItem(item)
   780→
